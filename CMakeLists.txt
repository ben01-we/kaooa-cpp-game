cmake_minimum_required(VERSION 3.16)
project(Kaooa LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This native desktop edition requires Windows.")
endif()
add_executable(Kaooa WIN32 main.cpp)
target_compile_features(Kaooa PRIVATE cxx_std_17)
target_compile_definitions(Kaooa PRIVATE UNICODE _UNICODE)
target_link_libraries(Kaooa PRIVATE gdiplus user32 gdi32)
if(MSVC)
  set_property(TARGET Kaooa PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
